% Problem 1
\noindent {\bfseries Problem 1.} If any of the following polynomials factor, factor them completely:
	\begin{2enumerate}
	\item $x^2 - 10x - 24$
	\item $25x^3 - 4x$
	\item $x^2 - 7x - 6$
	\item $x^3 + 8$
	\item $4x^2 + 8x + 3$
	\item $x^2 + 1$
	\item $x^4 + x^3 - 2x^2$
	\item $3x^2 - x - 2$
	\item $x^3 - 2x^2 + 5x - 10$
	\item $32x^2 - 76x + 35$ [Hint: This has roots $\tfrac{5}{8}, \tfrac{7}{4}$.]
	\end{2enumerate}
